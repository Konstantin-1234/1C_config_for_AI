&НаКлиенте
Функция ПодготовитьПутьКаталога(Знач Путь)
    Если Путь = Неопределено Тогда
        Возврат "";
    КонецЕсли;
    Путь = СокрЛП(Путь);
    Если Путь = "" Тогда
        Возврат "";
    КонецЕсли;
    Путь = СтрЗаменить(Путь, "/", "\");
    Если СтрДлина(Путь) = 2 И Прав(Путь, 1) = ":" Тогда
        Путь = Путь + "\";
    КонецЕсли;
    Пока Прав(Путь, 1) = "\" И СтрДлина(Путь) > 3 Цикл
        Путь = Лев(Путь, СтрДлина(Путь) - 1);
    КонецЦикла;
    Возврат Путь;
КонецФункции

// Проверяет, что целевой файл можно записать. Возвращает Истина/Ложь,
// при отказе заполняет ТекстОшибки.
&НаКлиенте
Функция ПроверитьВозможностьЗаписиФайла(Знач ПутьФайла, ТекстОшибки)
    Если ПустаяСтрока(ПутьФайла) Тогда
        ТекстОшибки = "Не указан путь целевого файла.";
        Возврат Ложь;
    КонецЕсли;
    Файл = Новый Файл(ПутьФайла);
    // Если файла еще нет, это нормально. Проверяем только уже существующий путь.
    Если Файл.Существует() Тогда
        Если Файл.ЭтоКаталог() Тогда
            ТекстОшибки = "Путь целевого файла занят каталогом: " + ПутьФайла;
            Возврат Ложь;
        КонецЕсли;
    КонецЕсли;
    // Дополнительно проверяем каталог, в котором должен быть создан файл.
    КаталогНазначения = Файл.Путь;
    Если Прав(КаталогНазначения, 1) = "\" И СтрДлина(КаталогНазначения) > 3 Тогда
        КаталогНазначения = Лев(КаталогНазначения, СтрДлина(КаталогНазначения) - 1);
    КонецЕсли;
    Если НЕ ПустаяСтрока(КаталогНазначения) Тогда
        ФайлКаталога = Новый Файл(КаталогНазначения);
        Если НЕ ФайлКаталога.Существует() Тогда
            ТекстОшибки = "Каталог назначения не существует: " + КаталогНазначения;
            Возврат Ложь;
        КонецЕсли;
        Если НЕ ФайлКаталога.ЭтоКаталог() Тогда
            ТекстОшибки = "Путь назначения не является каталогом: " + КаталогНазначения;
            Возврат Ложь;
        КонецЕсли;
    КонецЕсли;
    Возврат Истина;

&НаКлиенте
Функция НайтиФайлКонфигурации(Знач КаталогВыгрузки)
    ПутьФайла = НайтиФайлПоИмениБезУчетаРегистра(КаталогВыгрузки, "Configuration.xml");
    Если ПутьФайла <> "" Тогда
        Возврат ПутьФайла;
    КонецЕсли;
    РодительскийКаталог = ПолучитьРодительскийКаталог(КаталогВыгрузки);
    Если РодительскийКаталог <> "" Тогда
        ФайлРодитель = Новый Файл(РодительскийКаталог);
        Если ФайлРодитель.Существует() Тогда
            Если ФайлРодитель.ЭтоКаталог() Тогда
                ПутьФайла = НайтиФайлПоИмениБезУчетаРегистра(РодительскийКаталог, "Configuration.xml");
                Если ПутьФайла <> "" Тогда
                    Возврат ПутьФайла;
                КонецЕсли;
            КонецЕсли;
        КонецЕсли;
    КонецЕсли;
    Возврат "";
КонецФункции

&НаКлиенте
Функция НайтиФайлПоИмениБезУчетаРегистра(Знач Каталог, Знач ИмяФайла)
    Если Каталог = "" Тогда
        Возврат "";
    КонецЕсли;
    НайденныеФайлы = НайтиФайлы(Каталог, "*", Ложь);
    ИмяФайлаНРег = НРег(ИмяФайла);
    Для Каждого Файл Из НайденныеФайлы Цикл
        Если Файл.Существует() Тогда
            Если Файл.ЭтоФайл() И НРег(Файл.Имя) = ИмяФайлаНРег Тогда
                Возврат Файл.ПолноеИмя;
            КонецЕсли;
        КонецЕсли;
    КонецЦикла;
    Возврат "";
КонецФункции

        Возврат "";
    КонецЕсли;
    ПозицияПоследнегоРазделителя = 0;
    Для Индекс = 1 По СтрДлина(Путь) Цикл
        Если Сред(Путь, Индекс, 1) = "\" Тогда
            ПозицияПоследнегоРазделителя = Индекс;
        КонецЕсли;
    КонецЦикла;
    Если ПозицияПоследнегоРазделителя <= 1 Тогда
        Возврат "";
    КонецЕсли;
    Родитель = Лев(Путь, ПозицияПоследнегоРазделителя - 1);
    Если СтрДлина(Родитель) = 2 И Прав(Родитель, 1) = ":" Тогда
        Родитель = Родитель + "\";
    КонецЕсли;
    Возврат Родитель;
КонецФункции

// Обход каталога: каждый файл классифицируется и, если проходит фильтр
// по префиксу, добавляется в соответствующий список.
&НаКлиенте
Процедура СобратьФайлыДляУпаковки(Знач КорневойКаталог, Знач Префикс, Знач ПутьКонфигурации, СписокМетаданные, СписокКод, СписокИнтерфейс)

    Для Каждого Файл Из НайденныеФайлы Цикл
        Если НЕ Файл.Существует() Тогда
            Продолжить;
        КонецЕсли;
        Если НЕ Файл.ЭтоФайл() Тогда
            Продолжить;
        КонецЕсли;
        // Configuration.xml добавляется в файл 1 отдельно, первым блоком.
        Если НРег(Файл.ПолноеИмя) = НРег(ПутьКонфигурации) Тогда

            ДобавитьФайлВСписок(СписокМетаданные, КорневойКаталог, Файл);
        ИначеЕсли Группа = "Код" Тогда
            ДобавитьФайлВСписок(СписокКод, КорневойКаталог, Файл);
        ИначеЕсли Группа = "Интерфейс" Тогда
            ДобавитьФайлВСписок(СписокИнтерфейс, КорневойКаталог, Файл);

// Классификация файла по первому совпадению сверху вниз (см. заголовок области).
&НаКлиенте
Функция КлассифицироватьФайл(Знач Файл)
    ИмяФайлаНРег = НРег(Файл.Имя);
    РасширениеНРег = НРег(Файл.Расширение);
    Если ИмяФайлаНРег = "configdumpinfo.xml" Тогда
        Возврат "Исключить";
    КонецЕсли;
    Если РасширениеНРег = ".bsl" Тогда
        Возврат "Код";
    КонецЕсли;
    Если ИмяФайлаНРег = "commandinterface.xml" Тогда
        Возврат "Интерфейс";
    КонецЕсли;
    // Путь окаймляется "\", чтобы ловить и вложенные папки и файл прямо в папке.
    ПутьНРег = "\" + НРег(Файл.Путь) + "\";
    Если РасширениеНРег = ".xml" И СтрНайти(ПутьНРег, "\forms\") > 0 Тогда
        Возврат "Интерфейс";
    КонецЕсли;
    Если ИмяФайлаНРег = "homepageworkarea.xml" Тогда
        Возврат "Интерфейс";
    КонецЕсли;
    Если ИмяФайлаНРег = "help.xml" Тогда
        Возврат "Исключить";
    КонецЕсли;
    Если СтрНайти(ПутьНРег, "\templates\") > 0 Тогда
        Возврат "Исключить";
    КонецЕсли;
    Если РасширениеНРег = ".xml" Тогда
        Возврат "Метаданные";
    КонецЕсли;
    Возврат "Исключить";
КонецФункции

// Фильтр по префиксу: ищется в имени файла (без расширения) и в именах папок пути.

Функция ФайлСоответствуетПрефиксу(Знач Файл, Знач Префикс)
    Если ПустаяСтрока(Префикс) Тогда
        Возврат Истина;
    КонецЕсли;

        Если НЕ ПустаяСтрока(Часть) И СтрНайти(Часть, ПрефиксНРег) > 0 Тогда
            Возврат Истина;
        КонецЕсли;
    КонецЦикла;
    Возврат Ложь;

// Подготовка блоков всех трёх файлов. В файле 1 первым блоком всегда
// Configuration.xml [Main Config]. Нумерация блоков в каждом файле своя, с единицы.

// Добавляет блок сжатого XML. РежимСжатия: "Метаданные" или "Интерфейс".

Процедура ДобавитьБлокBSЛ(Блоки, Знач Номер, Знач ОтносительныйПуть, Знач ПолныйПуть)

    Чтение.Закрыть();
    Если Содержимое = Неопределено Тогда
        Содержимое = "";
    КонецЕсли;
    // Удаляем BOM, если есть.
    Если Лев(Содержимое, 1) = Символ(65279) Тогда
        Содержимое = Сред(Содержимое, 2);
    КонецЕсли;
    Блок = Новый Структура("Номер, ОтносительныйПуть, Тип, ПолныйПуть, Содержимое",
        Номер, ОтносительныйПуть, "BSL", ПолныйПуть, Содержимое);
    Блоки.Добавить(Блок);

// Сжатие XML построчно (в выгрузке 1С один тег на строку):
//  1) читаем файл, убираем BOM и ВК;
//  2) идём по строкам: блоки из списка на удаление пропускаем целиком
//     (с учётом вложенности), пустые теги <Tag/> выбрасываем;
//  3) постобработка - вырезаем xmlns, version, uuid, configVersion.
&НаКлиенте
Функция СжатьXMLФайл(Знач ПолныйПуть, Знач Режим)
    Чтение = Новый ЧтениеТекста(ПолныйПуть, КодировкаТекста.UTF8);
    ВесьТекст = Чтение.Прочитать();
    Чтение.Закрыть();
    Если ВесьТекст = Неопределено Тогда
        ВесьТекст = "";
    КонецЕсли;
    // Удаляем BOM, если есть.
    Если Лев(ВесьТекст, 1) = Символ(65279) Тогда
        ВесьТекст = Сред(ВесьТекст, 2);
    КонецЕсли;
    // Оставляем только ПС как разделитель строк.
    ВесьТекст = СтрЗаменить(ВесьТекст, Символы.ВК, "");
    
    ИменаУдаления = ИменаЭлементовДляУдаления(Режим);
    ПрефиксыУдаления = ПрефиксыЭлементовДляУдаления(Режим);
    
    Строки = СтрРазделить(ВесьТекст, Символы.ПС);
    РезультатСтроки = Новый Массив;
    // ГлубинаПропуска > 0 означает, что мы внутри удаляемого блока.
    ГлубинаПропуска = 0;
    
    Для Каждого Строка Из Строки Цикл
        Строка = СокрЛП(Строка);
        Если Строка = "" Тогда
            Продолжить;
        КонецЕсли;
        
        // Внутри удаляемого блока просто отслеживаем вложенность.
        Если ГлубинаПропуска > 0 Тогда
            ГлубинаПропуска = ГлубинаПропуска + ИзменениеГлубины(Строка);
            Продолжить;
        КонецЕсли;
        
        ИмяЭлемента = ИмяЭлементаИзСтроки(Строка);
        
        // Элемент из списка на удаление.
        Если ЭлементПодлежитУдалению(ИмяЭлемента, ИменаУдаления, ПрефиксыУдаления) Тогда
            Если ТипСтрокиXML(Строка) = "Start" Тогда
                // Блок с потомками: пропускаем до закрывающего тега.
                ГлубинаПропуска = 1;
            КонецЕсли;
            // Empty и Inline удаляются тем, что мы их не записываем.
            Продолжить;
        КонецЕсли;
        
        // Пустые теги <Tag/> удаляем.
        Если ТипСтрокиXML(Строка) = "Empty" Тогда
            Продолжить;
        КонецЕсли;
        
        РезультатСтроки.Добавить(Строка);
    КонецЦикла;
    
    Результат = СтрСоединить(РезультатСтроки, Символы.ПС);
    Возврат УдалитьСлужебныеАтрибутыИзТекста(Результат);
КонецФункции

// Тип строки XML: "End" (</Tag>), "Empty" (<Tag/>),
// "Inline" (<Tag>текст</Tag>), иначе "Start" (<Tag>).
&НаКлиенте
Функция ТипСтрокиXML(Знач Строка)
    Если Лев(Строка, 2) = "</" Тогда
        Возврат "End";
    КонецЕсли;
    Если Прав(Строка, 2) = "/>" Тогда
        Возврат "Empty";
    КонецЕсли;
    Если СтрНайти(Строка, "</") > 0 Тогда
        Возврат "Inline";
    КонецЕсли;
    Возврат "Start";
КонецФункции

// Изменение глубины вложенности для строки (для обхода удаляемого блока).
&НаКлиенте
Функция ИзменениеГлубины(Знач Строка)
    Тип = ТипСтрокиXML(Строка);
    Если Тип = "Start" Тогда
        Возврат 1;
    ИначеЕсли Тип = "End" Тогда
        Возврат -1;
    КонецЕсли;
    Возврат 0;
КонецФункции

// Имя элемента из строки: текст после "<" (или "</") до первого
// пробела, ">" или "/".
&НаКлиенте
Функция ИмяЭлементаИзСтроки(Знач Строка)
    Если Лев(Строка, 1) <> "<" Тогда
        Возврат "";
    КонецЕсли;
    Начало = 2;
    Если Лев(Строка, 2) = "</" Тогда
        Начало = 3;
    КонецЕсли;
    КонецИмени = СтрДлина(Строка) + 1;
    Для Индекс = Начало По СтрДлина(Строка) Цикл
        ТекущийСимвол = Сред(Строка, Индекс, 1);
        Если ТекущийСимвол = " " Или ТекущийСимвол = ">" Или ТекущийСимвол = "/" Или ТекущийСимвол = Символы.Таб Тогда
            КонецИмени = Индекс;
            Прервать;
        КонецЕсли;
    КонецЦикла;
    Возврат Сред(Строка, Начало, КонецИмени - Начало);
КонецФункции

// Подлежит ли элемент удалению: по локальному имени или по префиксу.
&НаКлиенте
Функция ЭлементПодлежитУдалению(Знач ИмяЭлемента, Знач ИменаУдаления, Знач ПрефиксыУдаления)
    Если ИмяЭлемента = "" Тогда
        Возврат Ложь;
    КонецЕсли;
    ЛокальноеИмя = НРег(ЛокальноеИмяИзИмени(ИмяЭлемента));
    Для Каждого Имя Из ИменаУдаления Цикл
        Если ЛокальноеИмя = НРег(Имя) Тогда
            Возврат Истина;
        КонецЕсли;
    КонецЦикла;
    Префикс = НРег(ПрефиксИзИмени(ИмяЭлемента));
    Если Префикс <> "" Тогда
        Для Каждого ПрефиксУдаления Из ПрефиксыУдаления Цикл
            Если Префикс = НРег(ПрефиксУдаления) Тогда
                Возврат Истина;
            КонецЕсли;
        КонецЦикла;
    КонецЕсли;
    Возврат Ложь;
КонецФункции

// Имя без префикса: "v8:item" -> "item".

    Если Поз > 0 Тогда
        Возврат Сред(Имя, Поз + 1);
    КонецЕсли;
    Возврат Имя;
КонецФункции

// Префикс имени: "dcsset:settings" -> "dcsset".
&НаКлиенте
Функция ПрефиксИзИмени(Знач Имя)
    Поз = СтрНайти(Имя, ":");
    Если Поз > 0 Тогда
        Возврат Лев(Имя, Поз - 1);
    КонецЕсли;
    Возврат "";
КонецФункции

// Постобработка: удаляет xmlns / xmlns:префикс, version (кроме <?xml?>),
// все uuid и configVersion.
&НаКлиенте
Функция УдалитьСлужебныеАтрибутыИзТекста(Знач Текст)
    // xmlns и xmlns:префикс.
    Пока Истина Цикл
        Поз = СтрНайти(Текст, " xmlns");
        Если Поз = 0 Тогда
            Прервать;
        КонецЕсли;
        НовыйТекст = ВырезатьАтрибутПоПозиции(Текст, Поз);
        Если НовыйТекст = Текст Тогда
            Прервать;
        КонецЕсли;
        Текст = НовыйТекст;
    КонецЦикла;
    // version - только после объявления <?xml ...?>.
    ПозОкончанияОбъявления = СтрНайти(Текст, "?>");
    НачалоПоиска = ?(ПозОкончанияОбъявления > 0, ПозОкончанияОбъявления + 2, 1);
    Пока Истина Цикл
        ПозОтн = СтрНайти(Сред(Текст, НачалоПоиска), " version=""");
        Если ПозОтн = 0 Тогда
            Прервать;
        КонецЕсли;
        Поз = НачалоПоиска + ПозОтн - 1;
        НовыйТекст = ВырезатьАтрибутПоПозиции(Текст, Поз);
        Если НовыйТекст = Текст Тогда
            Прервать;
        КонецЕсли;
        Текст = НовыйТекст;
    КонецЦикла;
    // uuid.
    Пока Истина Цикл
        Поз = СтрНайти(Текст, " uuid=""");
        Если Поз = 0 Тогда
            Прервать;
        КонецЕсли;
        НовыйТекст = ВырезатьАтрибутПоПозиции(Текст, Поз);
        Если НовыйТекст = Текст Тогда
            Прервать;
        КонецЕсли;
        Текст = НовыйТекст;
    КонецЦикла;
    // configVersion.
    Пока Истина Цикл
        Поз = СтрНайти(Текст, " configVersion=""");
        Если Поз = 0 Тогда
            Прервать;
        КонецЕсли;
        НовыйТекст = ВырезатьАтрибутПоПозиции(Текст, Поз);
        Если НовыйТекст = Текст Тогда
            Прервать;
        КонецЕсли;
        Текст = НовыйТекст;
    КонецЦикла;
    Возврат Текст;

// Вырезает атрибут вида ' name="value"', начинающийся с позиции Поз

&НаКлиенте
Функция ВырезатьАтрибутПоПозиции(Знач Текст, Знач Поз)
    ПозКавычки1Отн = СтрНайти(Сред(Текст, Поз), """");

    Если ПозКавычки2Отн = 0 Тогда
        Возврат Текст;
    КонецЕсли;
    ПозКавычки2 = ПозКавычки1 + ПозКавычки2Отн;

////////////////////////////////////////////////////////////////////////////////
//                        ЗАГРУЗКА ФАЙЛА В GITHUB                             //
////////////////////////////////////////////////////////////////////////////////

&НаКлиенте
Процедура ЗагрузитьФайлВGitHub(ПутьКФайлуЛокально, ИмяФайлаВРепозитории)
	
	// 1. Проверка существования файла на локальном диске
	Файл = Новый Файл(ПутьКФайлуЛокально);
	Если НЕ Файл.Существует() Тогда
		ПоказатьПредупреждение(, "Файл не найден:" + Символы.ПС + ПутьКФайлуЛокально);
		Возврат;
	КонецЕсли;
	
	Настройки = Новый Структура;
	Настройки.Вставить("Логин",ЭтаФорма.GitHub_Логин);
	Настройки.Вставить("Репозиторий",ЭтаФорма.GitHub_Репозиторий);
	Настройки.Вставить("Ветка",ЭтаФорма.GitHub_Ветка);
	Настройки.Вставить("Токен",ЭтаФорма.GitHub_Токен);
	// Защита от пустых настроек
	Если ПустаяСтрока(Настройки.Логин) Или ПустаяСтрока(Настройки.Репозиторий) Или ПустаяСтрока(Настройки.Токен) Тогда
		ПоказатьПредупреждение(, "Не заполнены настройки GitHub."
			+ Символы.ПС + "Логин: [" + Настройки.Логин + "]"
			+ Символы.ПС + "Репозиторий: [" + Настройки.Репозиторий + "]"
			+ Символы.ПС + "Ветка: [" + Настройки.Ветка + "]"
			+ Символы.ПС + "Токен: [" + Лев(Настройки.Токен, 7) + "...]");
		Возврат;
	КонецЕсли;
	
	// 2. Читаем файл и кодируем в Base64 (требование GitHub API)
	ДвоичныеДанные = Новый ДвоичныеДанные(ПутьКФайлуЛокально);
	СодержимоеBase64 = Base64Строка(ДвоичныеДанные);
	
	// 1С вставляет в Base64 переносы строк — строгий декодер GitHub их не принимает.
	// Убираем переносы (ПС и ВК), оставляем сплошную строку.
	СодержимоеBase64 = СтрЗаменить(СодержимоеBase64, Символы.ПС, "");
	СодержимоеBase64 = СтрЗаменить(СодержимоеBase64, Символы.ВК, "");
	
	// 3. Получаем SHA текущего файла на GitHub (для обновления; если файла нет — "")
	SHAФайла = GitHubПолучитьSHA(Настройки, ИмяФайлаВРепозитории);
	
	// 4. Формируем JSON-тело запроса
	ТелоЗапроса = Новый Структура;
	ТелоЗапроса.Вставить("message", "Автообновление из 1С " + Формат(ТекущаяДата(), "ДЛФ=DDT"));
	ТелоЗапроса.Вставить("content", СодержимоеBase64);
	ТелоЗапроса.Вставить("branch", Настройки.Ветка);
	Если SHAФайла <> "" Тогда
		ТелоЗапроса.Вставить("sha", SHAФайла);
	КонецЕсли;
	
	ЗаписьJSON = Новый ЗаписьJSON;
	ЗаписьJSON.УстановитьСтроку();
	ЗаписатьJSON(ЗаписьJSON, ТелоЗапроса);
	ТелоСтрокой = ЗаписьJSON.Закрыть();
	
	// 5. PUT-запрос к GitHub API (создание или обновление файла)
	Соединение = GitHubСоединение();
	Запрос = Новый HTTPЗапрос(GitHubПутьКФайлу(Настройки, ИмяФайлаВРепозитории));
	GitHubЗаполнитьЗаголовки(Запрос, Настройки);
	Запрос.Заголовки.Вставить("Content-Type", "application/json");
	Запрос.УстановитьТелоИзСтроки(ТелоСтрокой, КодировкаТекста.UTF8);
	
	Попытка
		Ответ = Соединение.Записать(Запрос);
	Исключение
		ПоказатьПредупреждение(, "Не удалось соединиться с GitHub:" + Символы.ПС + ОписаниеОшибки());
		Возврат;
	КонецПопытки;
	
	// 6. Разбор результата (200 — обновлен, 201 — создан)
	Если Ответ.КодСостояния = 200 Или Ответ.КодСостояния = 201 Тогда
		ПоказатьПредупреждение(, "Готово: файл """ + ИмяФайлаВРепозитории + """ обновлен на GitHub.");
	Иначе
		ПоказатьПредупреждение(, "Ошибка GitHub (код " + Ответ.КодСостояния + "):"
			+ Символы.ПС + "URL: https://api.github.com" + Запрос.АдресРесурса
			+ Символы.ПС + Ответ.ПолучитьТелоКакСтроку());
	КонецЕсли;
	
КонецПроцедуры

// Возвращает SHA файла на GitHub (нужен для обновления).
// Если файла еще нет (404) — возвращает пустую строку.
&НаКлиенте
Функция GitHubПолучитьSHA(Настройки, ИмяФайлаВРепозитории)
	
	SHA = "";
	
	Соединение = GitHubСоединение();
	Запрос = Новый HTTPЗапрос(GitHubПутьКФайлу(Настройки, ИмяФайлаВРепозитории) + "?ref=" + Настройки.Ветка);
	GitHubЗаполнитьЗаголовки(Запрос, Настройки);
	
	Попытка
		Ответ = Соединение.Получить(Запрос);
	Исключение
		Возврат "";
	КонецПопытки;
	
	Если Ответ.КодСостояния = 200 Тогда
		ЧтениеJSON = Новый ЧтениеJSON;
		ЧтениеJSON.УстановитьСтроку(Ответ.ПолучитьТелоКакСтроку());
		ДанныеОтвета = ПрочитатьJSON(ЧтениеJSON);
		ЧтениеJSON.Закрыть();
		SHA = ДанныеОтвета.sha;
	КонецЕсли;
	
	Возврат SHA;
