Процедура УдалитьДобавленныеКоманды()
	
	// Для корректного удаления пустых групп после кнопок,
	// запоминаем имена всех динамически добавленных групп.
	// Создадим соответствие для хранения уникальных имен групп.
	УникальныеИменаГрупп = Новый Соответствие;
	
	// Проходим по всем ранее добавленным командам и удаляем соответствующие элементы интерфейса
	// Используем итерацию с конца, чтобы избежать проблем с изменением исходной таблицы во время итерации
	
	// Получаем исходное количество строк
	ИсходноеКоличество = ДобавленныеКоманды.Количество();
	
	// Итерируемся от последней строки к первой (от большего индекса к меньшему)
	// Индексация строк в ТаблицеЗначений начинается с 0
	Для Счетчик = 0 По ИсходноеКоличество - 1 Цикл
		// Получаем строку, начиная с конца
		// Индекс строки для удаления = (ИсходноеКоличество - 1) - Счетчик
		ИндексДляУдаления = (ИсходноеКоличество - 1) - Счетчик;
		ТекущаяСтрока = ДобавленныеКоманды[ИндексДляУдаления];
		
		ИмяКнопки = ТекущаяСтрока.ИмяКнопки;
		// Удаляем команду, связанную с кнопкой, из коллекции команд формы
		// Для этого сначала нужно получить имя команды по имени кнопки
		ИмяКоманды = Элементы[ИмяКнопки].ИмяКоманды;
		Команды.Удалить(Команды[ИмяКоманды]);
		
		// Удаляем кнопку из элементов формы
		Элементы.Удалить( Элементы[ИмяКнопки]);
		
		// Запоминаем имя группы, к которой принадлежала кнопка, для последующего удаления пустой группы
		ИмяГруппыФормы = ТекущаяСтрока.ИмяГруппыФормы;
		// Проверяем, что имя группы заполнено и еще не было добавлено в соответствие
		// Проверка наличия ключа осуществляется с помощью метода Получить()
		Если ЗначениеЗаполнено(ИмяГруппыФормы) И УникальныеИменаГрупп.Получить(ИмяГруппыФормы) = Неопределено Тогда
			УникальныеИменаГрупп.Вставить(ИмяГруппыФормы, Истина); // Значение произвольное
		КонецЕсли;
		
		// Удаляем информацию о кнопке из реквизита формы (таблицы значений)
		// Это нужно делать *до* проверки необходимости удаления группы, чтобы не учитывать её
		ДобавленныеКоманды.Удалить(ТекущаяСтрока);
		
	КонецЦикла;
	
	// После удаления всех кнопок, удаляем пустые группы.
	// Проходимся по списку уникальных имен групп, которые были затронуты.
	Для Каждого КлючИзСоответствия Из УникальныеИменаГрупп Цикл
		ИмяГруппыФормы = КлючИзСоответствия.Ключ;
		
		// Проверяем, остались ли подчинённые элементы у этой группы в элементах формы
		// Получаем сам элемент группы из элементов формы по имени
		ЭлементГруппы = Элементы[ИмяГруппыФормы];
		
		// Убеждаемся, что элемент существует и является группой
		Если ТипЗнч(ЭлементГруппы) = Тип("ГруппаФормы") Тогда
			// Получаем коллекцию 
			//Подчиненные = ЭлементГруппы.ПодчиненныеЭлементы; //Ошибка. попытка переопределить предопределенный элемент
			// Считаем количество подчинённых элементов
			КоличествоПодчиненных = ЭлементГруппы.ПодчиненныеЭлементы.Количество();
			
			// Если подчинённых элементов нет, удаляем группу и связанную с ней команду
			Если КоличествоПодчиненных = 0 Тогда
				// Удаляем команду, связанную с группой
				// Для этого получаем имя команды из элемента группы
				//ИмяКомандыГруппы = ЭлементГруппы.ИмяКоманды;  //Тупой ИИ. Решил, что у группы кнопок должна быть команда
				//Команды.Удалить(ИмяКомандыГруппы);
				
				// Удаляем группу из элементов формы

Функция ПолучитьСостояние(ВидКарточки = Неопределено, Отбор = Неопределено, МенеджерВременныхТаблиц = Неопределено) Экспорт
	
	// 0. Инициализация менеджера
	Если Не ТипЗнч(МенеджерВременныхТаблиц) = Тип("МенеджерВременныхТаблиц") Тогда
		МенеджерВременныхТаблиц = Новый МенеджерВременныхТаблиц;
	КонецЕсли;
	
	// 1. Разбор отбора
	МассивКарточек = Новый Массив;
	ОтборыПоРеквизитам = Новый Соответствие;
	Если Отбор <> Неопределено Тогда
		Для Каждого КлючЗнач Из Отбор Цикл
			Если КлючЗнач.Ключ = "Карточка" И ТипЗнч(КлючЗнач.Значение) = Тип("СправочникСсылка._Карточки") Тогда
				Если МассивКарточек.Найти(КлючЗнач.Значение) = Неопределено Тогда
					МассивКарточек.Добавить(КлючЗнач.Значение);
				КонецЕсли;
			ИначеЕсли КлючЗнач.Ключ = "Карточки" И ТипЗнч(КлючЗнач.Значение) = Тип("Массив") Тогда
				Для Каждого Карта Из КлючЗнач.Значение Цикл
					Если МассивКарточек.Найти(Карта) = Неопределено Тогда
						МассивКарточек.Добавить(Карта);
					КонецЕсли;
				КонецЦикла;
			ИначеЕсли ТипЗнч(КлючЗнач.Ключ) = Тип("ПланВидовХарактеристикСсылка._ВидыРеквизитов") Тогда
				ОтборыПоРеквизитам.Вставить(КлючЗнач.Ключ, КлючЗнач.Значение);
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;
	
	Если ВидКарточки = Неопределено И МассивКарточек.Количество() > 0 Тогда
		ВидКарточки = МассивКарточек[0].ВидКарточки;
	КонецЕсли;
	Если ВидКарточки = Неопределено Тогда
		ВызватьИсключение "Не задан и не может быть определен параметр ВидКарточки.";
	КонецЕсли;
	
	// 2.1 Получение основных реквизитов карточки
	ПолучитьСостояние_ПодготовитьМенеджерВременныхТаблиц(ВидКарточки, МассивКарточек, ОтборыПоРеквизитам, МенеджерВременныхТаблиц);

	ЗапросСохранения0 = Новый Запрос;
	ЗапросСохранения0.МенеджерВременныхТаблиц = МенеджерВременныхТаблиц;
	ЗапросСохранения0.Текст = "

			ЗапросСсылок = Новый Запрос;
			ЗапросСсылок.МенеджерВременныхТаблиц = МенеджерВременныхТаблиц;
			ЗапросСсылок.Текст = "
			|ВЫБРАТЬ РАЗЛИЧНЫЕ
			|	Знач.ЗначениеРеквизита КАК СсылкаНаКарточку,
			|	Карточка.ВидКарточки КАК ВидКарточки

			|ИЗ
			|	" + ИмяТабПредыдущего + " КАК Знач
			|	ВНУТРЕННЕЕ СОЕДИНЕНИЕ " + ИмяТабСтруктурыПредыдущего + " КАК Струк
			|		ПО Знач.ВидРеквизита = Струк.ВидРеквизита
			|	ВНУТРЕННЕЕ СОЕДИНЕНИЕ Справочник._Карточки КАК Карточка
			|		ПО Знач.ЗначениеРеквизита = Карточка.Ссылка

			|	ТИПЗНАЧЕНИЯ(Знач.ЗначениеРеквизита) = ТИП(СправочникСсылка._Карточки)

			ЗапросСсылок.Выполнить();
			
			// Проверка наличия ссылок без двойного выполнения
			ЗапросПроверки = Новый Запрос;
			ЗапросПроверки.МенеджерВременныхТаблиц = МенеджерВременныхТаблиц;
			ЗапросПроверки.Текст = "ВЫБРАТЬ ПЕРВЫЕ 1 СсылкаНаКарточку ИЗ ВТ_СсылкиДляУровня";
			Если ЗапросПроверки.Выполнить().Пустой() Тогда
				ЗапросОчСсылки = Новый Запрос; ЗапросОчСсылки.МенеджерВременныхТаблиц = МенеджерВременныхТаблиц;
				ЗапросОчСсылки.Текст = "УНИЧТОЖИТЬ ВТ_СсылкиДляУровня";
				ЗапросОчСсылки.Выполнить();
				Прервать;
			КонецЕсли;
			
			// Группируем найденные ссылки по ВидамКарточек
			КартаВидов = Новый Соответствие;
			ВыборкаСсылок = ЗапросСсылок.Выполнить().Выбрать();
			Пока ВыборкаСсылок.Следующий() Цикл
				Вид = ВыборкаСсылок.ВидКарточки;
				Ссылка = ВыборкаСсылок.СсылкаНаКарточку;

					КартаВидов.Вставить(Вид, Новый Массив);

				КартаВидов.Получить(Вид).Добавить(Ссылка);
			КонецЦикла;
			
			Если КартаВидов.Количество() = 0 Тогда
				Прервать;
			КонецЕсли;
			
			// Создаем целевые таблицы для текущего уровня (пустые)
			ИмяТабТекущего = "ВТ_Значения_Ур" + Уровень;
			ИмяТабСтруктурыТекущего = "ВТ_Реквизиты_Ур" + Уровень;
			ЗапросСоздания = Новый Запрос; ЗапросСоздания.МенеджерВременныхТаблиц = МенеджерВременныхТаблиц;

			|ВЫБРАТЬ * ПОМЕСТИТЬ " + ИмяТабСтруктурыТекущего + " ИЗ " + ИмяТабСтруктурыПредыдущего + " ГДЕ ЛОЖЬ;";
			ЗапросСоздания.Выполнить();
			МассивТаблицЗначений.Добавить(ИмяТабТекущего);
			
			// Загружаем данные для каждого вида вложенной карточки
			Для Каждого ЭлементКарты Из КартаВидов Цикл
				// Вызываем подготовку. Фильтр видов реквизитов можно передать явно, если нужно ограничить вложенную структуру.
				ПолучитьСостояние_ПодготовитьМенеджерВременныхТаблиц(ЭлементКарты.Ключ, ЭлементКарты.Значение, Новый Соответствие, МенеджерВременныхТаблиц, Неопределено);

// Функция "ПолучитьСостояние"
// Возвращает текущие значения реквизитов Карточек соответствующие условиям отбора, 
// и временные таблицы с этими данными в МенеджерВременныхТаблиц.
// Возвращает массив структур с полями:
//   Карточка, ВидРеквизита, УИД_Элемента, ЗначениеРеквизита, Хранение.
//
// Параметры:
//  "Отбор" - Соответствие. Содержит условия отбора.
//  "ФильтрВидовРеквизитов" - Массив. Содержит ссылки на ПланВидовХарактеристик._ВидыРеквизитов.
//     Если задан, в результат попадут только значения указанных реквизитов.
//  "МенеджерВременныхТаблиц" - МенеджерВременныхТаблиц.
//  "Рекурсия" - Число. Разрешённая глубина рекурсии для получения вложенных реквизитов (по умолчанию 3).
Функция ПолучитьСостояние(ЗНАЧ Отбор = Неопределено, ЗНАЧ ФильтрВидовРеквизитов = Неопределено, МенеджерВременныхТаблиц = Неопределено, ЗНАЧ Рекурсия = 3) Экспорт
	// 1. Подготовка Менеджера Временных Таблиц
	Если Не ТипЗнч(МенеджерВременныхТаблиц) = Тип("МенеджерВременныхТаблиц") Тогда
		МенеджерВременныхТаблиц = Новый МенеджерВременныхТаблиц;
	КонецЕсли;

	// 2. Инициализация переменных
	МассивКарточек = Новый Массив;
	ОтборыПоРеквизитам = Новый Соответствие;
	ВидКарточки = Неопределено;

	// 3.1 Обработка параметра "Отбор"
	Если Отбор <> Неопределено Тогда
		Для Каждого КлючЗнач Из Отбор Цикл
			Если КлючЗнач.Ключ = "Карточка" 
				И ТипЗнч(КлючЗнач.Значение) = Тип("СправочникСсылка._Карточки") Тогда
				Если МассивКарточек.Найти(КлючЗнач.Значение) = Неопределено Тогда
					МассивКарточек.Добавить(КлючЗнач.Значение);
				КонецЕсли;
			ИначеЕсли КлючЗнач.Ключ = "Карточки" 
				И ТипЗнч(КлючЗнач.Значение) = Тип("Массив") Тогда
				Для Каждого Карточка Из КлючЗнач.Значение Цикл
					Если МассивКарточек.Найти(Карточка) = Неопределено Тогда
						МассивКарточек.Добавить(Карточка);
					КонецЕсли;
				КонецЦикла;
			ИначеЕсли КлючЗнач.Ключ = "ВидКарточки" Тогда 
				ВидКарточки = КлючЗнач.Значение;
			ИначеЕсли ТипЗнч(КлючЗнач.Ключ) = Тип("ПланВидовХарактеристикСсылка._ВидыРеквизитов") Тогда
				ОтборыПоРеквизитам.Вставить(КлючЗнач.Ключ, КлючЗнач.Значение);
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;

	// 4. Определение ВидКарточки
	// если вид карточки не указан, но установлен отбор по картачкам, буду использовать вид карточки из первой карточки.
	// Вид карточки должен быть указан явно, т.к. на основании вида карточки определяются реквизиты которые нужно извлекать
	Если ВидКарточки = Неопределено И МассивКарточек.Количество() > 0 Тогда
		ВидКарточки = МассивКарточек[0].ВидКарточки;
	КонецЕсли;

	Если ВидКарточки = Неопределено Тогда
		ВызватьИсключение "Не задан и не может быть определен параметр ВидКарточки в отборе или по карточкам.";
	КонецЕсли;

	// 5. Формирование текста запроса
	ПараметрыЗапроса = Новый Структура;
	ПараметрыЗапроса.Вставить("ВидКарточки", ВидКарточки);

	ТекстЗапросаНаУдаление = "";

	// 5.01 ВТ_Реквизиты (шаблон вида карточки)
	// Левое соединение с этой таблицей гарантирует, что в результат попадут ТОЛЬКО реквизиты, 
	// описанные в ТЧ "Реквизиты" текущего ВидаКарточки.
	// 5.01 
	// ВТ_Реквизиты - список реквизитов значения, которых нужно вернуть в результате
	// ВТ_Реквизиты_Вычисляемые - список реквизитов значения, которых нужно вычислить. в этой таблице присутствуют реквизиты, которые будут использоваться в итерациях

		|ПОМЕСТИТЬ ВТ_Реквизиты
		|ИЗ

		|ГДЕ
		|	ВидРеквизита В (&МассивВидовРеквизитов)
		|;
		|
		|////////////////////////////////////////////////////////////////////////////////
		|ВЫБРАТЬ
		|	ВидКарточки КАК ВидКарточки,

		|	Хранение КАК Хранение
		// продумать логику, походу эта ВТ должна называться ВТ_Реквизиты_Для_Вычисления
		|ПОМЕСТИТЬ ВТ_Реквизиты_Вычисляемые
		|ИЗ

		|	ИЛИ ВидРеквизита В (ВЫБРАТЬ ВТ_Реквизиты.Хранение ИЗ ВТ_Реквизиты))
		|;				
		|";

	Если ФильтрВидовРеквизитов <> Неопределено 
		И ТипЗнч(ФильтрВидовРеквизитов) = Тип("Массив") 
		И ФильтрВидовРеквизитов.Количество() > 0

	КонецЕсли;
	ТекстЗапросаНаУдаление = ТекстЗапросаНаУдаление + "
		|УНИЧТОЖИТЬ ВТ_Реквизиты_Все;
		|УНИЧТОЖИТЬ ВТ_Реквизиты_Вычисляемые;";

	ВТ_ТаблицаОтобраныхКарточек = "ВТ_Пустая";

	// 5.11 Фильтр по явно заданным карточкам
	Если МассивКарточек.Количество() > 0 Тогда
		Условие_МассивКарточек = " И Карточка В (&МассивКарточек)";
		ПараметрыЗапроса.Вставить("МассивКарточек", МассивКарточек);
		ТекстЗапросаОтобранныхКарточек = ТекстЗапросаОтобранныхКарточек + "
		|ВЫБРАТЬ
		|	_Карточки.Ссылка КАК Карточка
		|ПОМЕСТИТЬ ВТ_Карточки
		|ИЗ
		|	Справочник._Карточки КАК _Карточки
		|ГДЕ
		|	_Карточки.Ссылка В (&МассивКарточек)

		ВТ_ТаблицаОтобраныхКарточек = "ВТ_Карточки";
